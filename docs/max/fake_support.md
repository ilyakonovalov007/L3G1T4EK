        </button>

    <div v-if="!showFeedback" style="display: flex; flex-direction: column; gap: 8px;">
      <button @click="handleChoice('reply')" style="padding: 12px; background: var(--vp-c-bg); border: 1px solid #3b82f6; color: #3b82f6; border-radius: 8px; font-weight: 600; cursor: pointer; text-align: left;">
        💬 Ответить / Перейти по ссылке

<script setup>
import { ref, computed } from 'vue'

const currentStep = ref(0)
const userHealth = ref(100)
const showFeedback = ref(false)
const selectedChoice = ref('')

const steps = [
  {
    sender: 'Служба безопасности МАКС',
    avatar: '🤖',
    text: 'Внимание! Ваш профиль МАКС заблокируют через 2 часа из-за подозрительной активности. Чтобы сохранить доступ к аккаунту и привязанным Госуслугам, срочно подтвердите личность на нашем портале проверки:',
    link: 'https://max-support-help.ru/auth',
    feedback: {
      reply: '❌ Ужасно! Вы перешли по фишинговой ссылке и ввели данные аккаунта. Мошенники мгновенно перехватили контроль. Минус 40% защиты.',
      ignore: '⚠️ Рискованно. Вы просто закрыли чат. Но злоумышленник всё ещё висит в списке контактов и может продолжить атаку на невнимательных родственников.',
      block: '✅ Идеально! Вы сразу распознали поддельный домен (max-support-help.ru) и попытку запугать вас ограничением по времени ("через 2 часа"). Отличная реакция!'
    }
  },
  {
    sender: 'Налоговый инспектор (МАКС) ✔️',
    avatar: '🤵',
    text: 'Здравствуйте! Я представляю ведомственную техподдержку. В вашей анкете обнаружена ошибка в реквизитах, из-за чего заблокирована выплата компенсаций за 2025 год. Мы отправили вам проверочный код в SMS. Напишите его сюда, чтобы мы исправили анкету автоматически.',
    link: '',
    feedback: {
      reply: '❌ Катастрофа! СМС-код — это ключ к изменению пароля от ваших Госуслуг. Вы отдали его мошенникам, они зашли в ваш настоящий кабинет. Минус 50% защиты.',
      ignore: '⚠️ Не совсем верно. Игнорирование спасает конкретно ваш аккаунт, но мошенника лучше полностью блокировать, чтобы система забанила его быстрее.',
      block: '✅ Абсолютно верно! Настоящие ведомства никогда не требуют СМС-коды в личной переписке. Эмодзи галочки в имени (✔️) — это лишь дешёвый трюк для обмана.'
    }
  },
  {
    sender: 'Техподдержка МАКС-Защита',
    avatar: '🛡️',
    text: 'Важная информация для пользователей платформы МАКС! В связи с участившимися хакерскими атаками, всем пользователям необходимо обновить внутренний сертификат безопасности. Скачайте официальный защитный модуль max_security_update.apk и запустите его на вашем смартфоне для защиты Госуслуг.',
    link: '',
    feedback: {
      reply: '❌ Критическая ошибка! Под видом "сертификата" или "обновления" мошенники присылают троян (вирус), который получит доступ ко всем вашим СМС, паролям и банковским приложениям. Минус 50% защиты.',
      ignore: '⚠️ Пассивная позиция. Оставлять вредоносные файлы и ссылки в активных чатах нельзя — можно случайно нажать на них позже. Лучше заблокировать.',
      block: '✅ Великолепно! Никакие технические службы поддержки не рассылают приложения или файлы `.apk` через личные сообщения. Вы спасли свой телефон от вируса!'
    }
  }
]
const selectedChoiceText = computed(() => {
  if (selectedChoice.value === 'reply') return '💬 Я ответил на сообщение / перешел по ссылке'
  if (selectedChoice.value === 'ignore') return '⏳ Я решил проигнорировать этот чат'
  if (selectedChoice.value === 'block') return '🚫 Я заблокировал контакт и отправил жалобу'
  return ''
})

const feedbackTitle = computed(() => {
  if (selectedChoice.value === 'block') return '🟢 Правильные действия:'
  if (selectedChoice.value === 'ignore') return '🟡 Комментарий:'
  return '🔴 Допущена критическая ошибка:'
})

const feedbackColor = computed(() => {
  if (selectedChoice.value === 'block') return '#10b981'
  if (selectedChoice.value === 'ignore') return '#f59e0b'
  return '#ef4444'
})

const handleChoice = (type) => {
  selectedChoice.value = type
  showFeedback.value = true
  
  if (type === 'reply') {
    if (currentStep.value === 1) userHealth.value -= 50
    else userHealth.value -= 40
    if (userHealth.value < 0) userHealth.value = 0
  }
}

const nextStep = () => {
  showFeedback.value = false
  selectedChoice.value = ''
  currentStep.value++
}

const restartQuest = () => {
  currentStep.value = 0
  userHealth.value = 100
  showFeedback.value = false
  selectedChoice.value = ''
}

const finishLesson = () => {
  if (typeof window !== 'undefined') {
    localStorage.setItem('lesson_max_support_completed', 'true')
    window.location.href = '/max-info'
  }
}
</script>
